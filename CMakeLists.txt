cmake_minimum_required(VERSION 3.16)
project(g1_apple_throw LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(Eigen3 REQUIRED)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  set(ARCH aarch64)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
  set(ARCH x86_64)
else()
  message(FATAL_ERROR "Supported targets: aarch64 and x86_64")
endif()
set(SDK ${CMAKE_SOURCE_DIR}/thirdparty/unitree_sdk2)
set(ORT ${CMAKE_SOURCE_DIR}/thirdparty/onnxruntime/${ARCH})
add_library(throw_core INTERFACE)
target_include_directories(throw_core INTERFACE include ${ORT}/include)
target_link_libraries(throw_core INTERFACE yaml-cpp Eigen3::Eigen ${ORT}/lib/libonnxruntime.so.1.22.0 Threads::Threads)
add_executable(throw_self_test src/self_test.cpp)
target_link_libraries(throw_self_test PRIVATE throw_core)
add_executable(g1_throw src/controller.cpp)
target_include_directories(g1_throw PRIVATE ${SDK}/include ${SDK}/thirdparty/include ${SDK}/thirdparty/include/ddscxx)
target_link_libraries(g1_throw PRIVATE throw_core ${SDK}/lib/${ARCH}/libunitree_sdk2.a
  ${SDK}/thirdparty/lib/${ARCH}/libddscxx.so ${SDK}/thirdparty/lib/${ARCH}/libddsc.so dl rt)
foreach(TARGET throw_self_test g1_throw)
 target_compile_options(${TARGET} PRIVATE -O2 -Wall -Wextra)
 set_target_properties(${TARGET} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/bin/${ARCH}
   BUILD_RPATH "${ORT}/lib;${SDK}/thirdparty/lib/${ARCH}"
   INSTALL_RPATH "$ORIGIN/../../thirdparty/onnxruntime/${ARCH}/lib;$ORIGIN/../../thirdparty/unitree_sdk2/thirdparty/lib/${ARCH}"
   BUILD_WITH_INSTALL_RPATH TRUE)
endforeach()
enable_testing()
add_test(NAME deployment_parity COMMAND throw_self_test ${CMAKE_SOURCE_DIR})
option(BUILD_LOCAL_DDS_FIXTURE "Build offline loopback-only transport fixture" OFF)
if(BUILD_LOCAL_DDS_FIXTURE)
 add_executable(local_dds_fixture src/local_dds_fixture.cpp)
 get_target_property(RUNTIME_INCLUDES g1_throw INCLUDE_DIRECTORIES)
 get_target_property(RUNTIME_LINKS g1_throw LINK_LIBRARIES)
 target_include_directories(local_dds_fixture PRIVATE ${RUNTIME_INCLUDES})
 target_link_libraries(local_dds_fixture PRIVATE ${RUNTIME_LINKS})
 set_target_properties(local_dds_fixture PROPERTIES BUILD_RPATH "${ORT}/lib;${SDK}/thirdparty/lib/${ARCH}")
endif()
